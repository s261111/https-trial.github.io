let current = 0;
let userAnswer = [];
let realAge = 0;

function startTest(){
    realAge = Number(document.getElementById('realAge').value);
    if(!realAge || realAge<10 || realAge>80){
        alert("请输入10-80之间的真实年龄");
        return;
    }
    document.getElementById('start-page').classList.add('hidden');
    document.getElementById('test-page').classList.remove('hidden');
    renderQuestion();
}

function renderQuestion(){
    const q = questionList[current];
    document.getElementById('current-num').innerText = current+1;
    document.getElementById('question-title').innerText = q.q;
    const box = document.getElementById('option-box');
    box.innerHTML = "";
    q.o.forEach((opt,idx)=>{
        const item = document.createElement('div');
        item.className = "option-item border border-gray-200 rounded-lg px-4 py-3 cursor-pointer hover:border-primary";
        if(userAnswer[current] === idx) item.classList.add('bg-gray-100','border-primary');
        item.innerText = opt;
        item.onclick = ()=>{
            userAnswer[current] = idx;
            renderQuestion();
        }
        box.appendChild(item);
    })
    const percent = ((current+1)/40)*100;
    document.getElementById('progress-bar').style.width = percent + "%";
    document.getElementById('next-btn').innerText = current === 39 ? "查看结果" : "下一题";
}

async function nextQuestion(){
    if(userAnswer[current] === undefined){
        alert("请选择一个答案");
        return;
    }
    if(current < 39){
        current++;
        renderQuestion();
    }else{
        // 提交，带上token在header
        try{
            const res = await fetch(`${API_BASE}/api/submit-test`,{
                method:"POST",
                headers:{
                    "Content-Type":"application/json",
                    "X-Request-Token": requestToken
                },
                body: JSON.stringify({
                    realAge,
                    answers: userAnswer
                })
            });
            const json = await res.json();
            if(json.code !== 0){
                alert(json.msg);
                return;
            }
            // 渲染后端返回结果
            const data = json.data;
            document.getElementById('test-page').classList.add('hidden');
            document.getElementById('result-page').classList.remove('hidden');
            document.getElementById('res-real').innerText = data.realAge;
            document.getElementById('res-mental').innerText = data.mentalAge;
            document.getElementById('diff-text').innerText = data.diffText;
            document.getElementById('analysis-text').innerText = data.analysis;
        }catch(e){
            alert("提交失败，网络异常");
        }
    }
}

async function restart(){
    // 重新测试，重新获取新token
    const res = await fetch(`${API_BASE}/api/get-token`);
    const json = await res.json();
    if(json.code === 0){
        requestToken = json.data.request_token;
    }
    current = 0;
    userAnswer = [];
    document.getElementById('result-page').classList.add('hidden');
    document.getElementById('start-page').classList.remove('hidden');
}
